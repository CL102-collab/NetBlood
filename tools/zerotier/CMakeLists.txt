cmake_minimum_required(VERSION 3.12)
project(netblood_zerotier_companion LANGUAGES C CXX)
if(NOT ZEROTIER_SDK_SOURCE)
    message(FATAL_ERROR "Run tools/zerotier/build.py, or set ZEROTIER_SDK_SOURCE to the patched, complete pinned SDK")
endif()
set(BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_HOST_SELFTEST OFF CACHE BOOL "" FORCE)
set(BUILD_HOST_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALLOW_INSTALL_TARGET OFF CACHE BOOL "" FORCE)
set(ZTS_DISABLE_CENTRAL_API ON CACHE BOOL "" FORCE)
add_compile_definitions(ZTS_STATIC=1 OMIT_JSON_SUPPORT=1)
add_subdirectory("${ZEROTIER_SDK_SOURCE}" sdk EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_executable(netblood-zerotier companion.cpp)
target_compile_features(netblood-zerotier PRIVATE cxx_std_17)
target_include_directories(netblood-zerotier PRIVATE "${ZEROTIER_SDK_SOURCE}/include" ../../source/build/include)
target_link_libraries(netblood-zerotier PRIVATE zt-static Threads::Threads)
if(WIN32)
    target_link_libraries(netblood-zerotier PRIVATE ws2_32 iphlpapi shlwapi)
endif()
install(TARGETS netblood-zerotier RUNTIME DESTINATION .)
